const isNative = !!(window.Capacitor && window.Capacitor.isNativePlatform && window.Capacitor.isNativePlatform());

// ═══════════════════════════════════════════════════════
//  File System Abstraction
// ═══════════════════════════════════════════════════════
const FS = {
  async pickDirectory() {
    if (isNative) { const r = await SafPlugin.pickDirectory(); return { ref: r.uri, name: r.name }; }
    if (!window.showDirectoryPicker) throw new Error('Browser does not support folder selection');
    const h = await window.showDirectoryPicker({ mode: 'readwrite' });
    return { ref: h, name: h.name };
  },
  async listWavFiles(folderRef) {
    if (isNative) {
      const r = await SafPlugin.listWavFiles({ uri: folderRef });
      const arr = Array.isArray(r.files) ? r.files : JSON.parse(JSON.stringify(r.files || []));
      return arr.filter(f => f.name.toLowerCase().endsWith('.wav')).sort((a,b) => a.name.localeCompare(b.name,undefined,{numeric:true})).map(f => ({ ref: f.uri, name: f.name }));
    }
    const files = [];
    for await (const e of folderRef.values()) { if (e.kind==='file' && e.name.toLowerCase().endsWith('.wav')) files.push({ref:e, name:e.name}); }
    return files.sort((a,b) => a.name.localeCompare(b.name,undefined,{numeric:true}));
  },
  async getPlaybackUrl(fileRef) {
    if (isNative) { const r = await SafPlugin.getFileForPlayback({uri:fileRef}); return window.Capacitor.convertFileSrc(r.path); }
    return URL.createObjectURL(await fileRef.getFile());
  },
  async readFileBuffer(fileRef) {
    if (isNative) { const r = await SafPlugin.readFileData({uri:fileRef}); return base64ToArrayBuffer(r.data); }
    return await (await fileRef.getFile()).arrayBuffer();
  },
  async writeFile(folderRef, fileName, arrayBuffer) {
    if (isNative) { const b64=arrayBufferToBase64(arrayBuffer); const r=await SafPlugin.writeFile({dirUri:folderRef,fileName,data:b64}); return {ref:r.uri, name:r.name||fileName}; }
    const h = await folderRef.getFileHandle(fileName,{create:true}); const w=await h.createWritable(); await w.write(arrayBuffer); await w.close(); return {ref:h, name:fileName};
  },
  async deleteFile(folderRef, fileName, fileRef) {
    if (isNative) { await SafPlugin.deleteFile({uri:fileRef}); } else { await folderRef.removeEntry(fileName); }
  },
  async getSavedFolders() {
    if (isNative) { const r=await SafPlugin.getSavedFolders(); const a=Array.isArray(r.folders)?r.folders:JSON.parse(JSON.stringify(r.folders||[])); return a.map(f=>({ref:f.uri,name:f.name})); }
    return await idbGetAllFolders();
  },
  async saveFolder(ref,name) { if (!isNative) return await idbSaveFolder(ref,name); return null; },
  async removeFolder(ref) { if (isNative) await SafPlugin.removeSavedFolder({uri:ref}); },
  async requestPermission(handle) {
    if (isNative) return true;
    const o={mode:'readwrite'}; if (await handle.queryPermission(o)==='granted') return true; return (await handle.requestPermission(o))==='granted';
  }
};

function arrayBufferToBase64(buf) { const b=new Uint8Array(buf); let r=''; for(let i=0;i<b.length;i+=32768){const c=b.subarray(i,Math.min(i+32768,b.length)); r+=String.fromCharCode.apply(null,c);} return btoa(r); }
function base64ToArrayBuffer(s) { const b=atob(s); const a=new Uint8Array(b.length); for(let i=0;i<b.length;i++) a[i]=b.charCodeAt(i); return a.buffer; }

// ═══════════════════════════════════════════════════════
//  FileCache — localStorage cache for file lists
// ═══════════════════════════════════════════════════════
const FileCache = {
  _key(name) { return 'fc_' + name; },
  save(folderName, files) {
    try {
      const d = files.map(f => ({ name: f.name, ref: isNative ? f.ref : null }));
      localStorage.setItem(this._key(folderName), JSON.stringify({ files: d, ts: Date.now() }));
    } catch(e) {}
  },
  load(folderName) {
    try { const r=localStorage.getItem(this._key(folderName)); if(!r) return null; return JSON.parse(r).files||null; } catch(e) { return null; }
  },
  remove(folderName) { try { localStorage.removeItem(this._key(folderName)); } catch(e) {} },
};

// ═══════════════════════════════════════════════════════
//  FavCache — in-memory + localStorage cache for fav file names
// ═══════════════════════════════════════════════════════
const FavCache = {
  _data: {},
  _loaded: {},

  async refresh(index) {
    const fav = favState.folders[index];
    if (!fav) { this._data[index] = new Set(); this._loaded[index] = true; return; }
    try {
      const files = await FS.listWavFiles(fav.ref);
      this._data[index] = new Set(files.map(f => f.name));
      this._persist(index);
    } catch(e) { this._data[index] = new Set(); }
    this._loaded[index] = true;
  },

  loadFromDisk(index) {
    const fav = favState.folders[index];
    if (!fav) { this._data[index] = new Set(); return; }
    try {
      const raw = localStorage.getItem('favc_' + index);
      this._data[index] = raw ? new Set(JSON.parse(raw)) : new Set();
      if (raw) this._loaded[index] = true;
    } catch(e) { this._data[index] = new Set(); }
  },

  has(index, fileName) { return this._data[index] ? this._data[index].has(fileName) : false; },
  add(index, fileName) { if (!this._data[index]) this._data[index] = new Set(); this._data[index].add(fileName); this._persist(index); },
  remove(index, fileName) { if (this._data[index]) this._data[index].delete(fileName); this._persist(index); },

  _persist(index) {
    try { localStorage.setItem('favc_' + index, JSON.stringify([...(this._data[index] || [])])); } catch(e) {}
  },

  invalidate(index) { this._loaded[index] = false; },
};

function idbOpen() {
  return new Promise((res,rej) => {
    const r=indexedDB.open('WavPlayerDB',2);
    r.onupgradeneeded=e => { const db=e.target.result; if(!db.objectStoreNames.contains('folders')) db.createObjectStore('folders',{keyPath:'id',autoIncrement:true}); if(!db.objectStoreNames.contains('favFolders')) db.createObjectStore('favFolders',{keyPath:'index'}); };
    r.onsuccess=e=>res(e.target.result); r.onerror=e=>rej(e.target.error);
  });
}
async function idbSaveFolder(h,n) { const db=await idbOpen(); return new Promise((res,rej)=>{const tx=db.transaction('folders','readwrite'); const r=tx.objectStore('folders').add({handle:h,name:n}); r.onsuccess=()=>res(r.result); tx.onerror=e=>rej(e.target.error);}); }
async function idbGetAllFolders() { const db=await idbOpen(); return new Promise((res,rej)=>{const r=db.transaction('folders','readonly').objectStore('folders').getAll(); r.onsuccess=()=>res(r.result.map(x=>({ref:x.handle,name:x.name,dbId:x.id}))); r.onerror=e=>rej(e.target.error);}); }
async function idbRemoveFolder(id) { const db=await idbOpen(); return new Promise((res,rej)=>{const tx=db.transaction('folders','readwrite'); tx.objectStore('folders').delete(id); tx.oncomplete=()=>res(); tx.onerror=e=>rej(e.target.error);}); }

async function doFavorite(index) {
  const folder=state.folders[state.activeFolderIdx];
  if (!folder||state.activeTrackIdx<0) { toast('Please select an audio file first','warning'); return; }
  const fav=favState.folders[index-1];
  if (!fav) { toast(`Please set favorite folder ${index} in the sidebar first`,'warning'); return; }
  const fileEntry=folder.files[state.activeTrackIdx]; const fi=index-1;
  try {
    if (FavCache.has(fi, fileEntry.name)) {
      const favFiles=await FS.listWavFiles(fav.ref); const ex=favFiles.find(f=>f.name===fileEntry.name);
      if (ex) await FS.deleteFile(fav.ref, ex.name, ex.ref);
      FavCache.remove(fi, fileEntry.name);

    } else {
      const buf=await FS.readFileBuffer(fileEntry.ref);
      await FS.writeFile(fav.ref, fileEntry.name, buf);
      FavCache.add(fi, fileEntry.name);
      toast(`Added to favorite folder ${index}`, `fav${index}`);

    updateFavButtonStates();
  } catch(e) { toast('Operation failed: '+e.message,'error'); }
}

async function initSavedFolders() {
  try {
    const saved = await FS.getSavedFolders();
    for (const item of saved) {
      const cached = FileCache.load(item.name);
      const folder = { ref:item.ref, name:item.name, files:[], dbId:item.dbId||null };
      if (isNative && cached && cached.length > 0) {
        folder.files = cached.map(f => ({ ref: f.ref, name: f.name }));
      } else if (!isNative) {
        try {
          const ok = await FS.requestPermission(item.ref);
          if (ok) {
            const files = await FS.listWavFiles(item.ref);
            folder.files = files;
            FileCache.save(item.name, files);
          }
        } catch(e) { console.warn('Scan failed', item.name, e); }
      }
      state.folders.push(folder);
    }
    renderFolders();
    if (state.folders.length > 0) {
      selectFolder(0);
      if (state.folders[0].files.length > 0) {
        playTrack(0);
      }
    }
  } catch(e) { console.warn('Init folders failed',e); }
}

async function bgRefreshFolder(idx) {
  const folder = state.folders[idx]; if(!folder) return;
  try {
    if (!isNative) { const ok=await FS.requestPermission(folder.ref); if(!ok) return; }
    const fresh = await FS.listWavFiles(folder.ref);
    const oldNames = folder.files.map(f=>f.name).join('\n');
    const newNames = fresh.map(f=>f.name).join('\n');

    if (oldNames === newNames) {
      const freshMap = new Map(fresh.map(f => [f.name, f.ref]));
      for (const f of folder.files) { const r = freshMap.get(f.name); if (r) f.ref = r; }
      FileCache.save(folder.name, folder.files);
    } else {
      const playingName = (state.activeFolderIdx === idx && state.activeTrackIdx >= 0)
        ? folder.files[state.activeTrackIdx]?.name : null;
      folder.files = fresh;
      FileCache.save(folder.name, fresh);
      if (playingName && state.activeFolderIdx === idx) {
        const ni = fresh.findIndex(f => f.name === playingName);
        if (ni >= 0) state.activeTrackIdx = ni;
      }
      if (state.activeFolderIdx === idx) renderTracks();
      renderFolders();
    }
  } catch(e) { console.warn('Refresh error',folder.name,e); }
}

async function manualRefreshFolder(idx, e) {

  await bgRefreshFolder(idx);
  if (btn) setTimeout(()=>btn.classList.remove('spinning'), 300);
  const f=state.folders[idx]; if(f) toast(`Refreshed "${f.name}" - ${f.files.length} files`,'success');
}

function parseWavHeader(buffer){const v=new DataView(buffer);const nc=v.getUint16(22,true);const sr=v.getUint32(24,true);const bps=v.getUint16(34,true);let o=12;while(o<buffer.byteLength-8){const id=String.fromCharCode(v.getUint8(o),v.getUint8(o+1),v.getUint8(o+2),v.getUint8(o+3));const sz=v.getUint32(o+4,true);if(id==='data')return{numChannels:nc,sampleRate:sr,bitsPerSample:bps,dataOffset:o+8,dataSize:sz};o+=8+sz;if(o%2!==0)o++;}throw new Error('data chunk not found');}
function createWavBuffer(sampleBytes,numChannels,sampleRate,bitsPerSample){const ds=sampleBytes.byteLength;const buf=new ArrayBuffer(44+ds);const v=new DataView(buf);const w=(off,s)=>{for(let i=0;i<s.length;i++)v.setUint8(off+i,s.charCodeAt(i));};const bps=bitsPerSample/8;w(0,'RIFF');v.setUint32(4,36+ds,true);w(8,'WAVE');w(12,'fmt ');v.setUint32(16,16,true);v.setUint16(20,1,true);v.setUint16(22,numChannels,true);v.setUint32(24,sampleRate,true);v.setUint32(28,sampleRate*numChannels*bps,true);v.setUint16(32,numChannels*bps,true);v.setUint16(34,bitsPerSample,true);w(36,'data');v.setUint32(40,ds,true);new Uint8Array(buf,44).set(new Uint8Array(sampleBytes));return buf;}

function convertPcmData(buffer,header,tgtCh,tgtRate,tgtBits){const{numChannels:sCh,sampleRate:sR,bitsPerSample:sB,dataOffset:dO,dataSize:dS}=header;if(sCh===tgtCh&&sR===tgtRate&&sB===tgtBits)return buffer.slice(dO,dO+dS);const sBps=sB/8,tBps=tgtBits/8,sFc=dS/(sBps*sCh),sv=new DataView(buffer);function rd(p,b){if(b===8)return(sv.getUint8(p)-128)/128;if(b===16)return sv.getInt16(p,true)/32768;if(b===24)return((sv.getUint8(p)|(sv.getUint8(p+1)<<8)|(sv.getInt8(p+2)<<16)))/8388608;if(b===32)return sv.getInt32(p,true)/2147483648;return 0;}
const sf=[];for(let c=0;c<sCh;c++)sf.push(new Float64Array(sFc));for(let i=0;i<sFc;i++){const fp=dO+i*sBps*sCh;for(let c=0;c<sCh;c++)sf[c][i]=rd(fp+c*sBps,sB);}
let cf;if(sCh===tgtCh)cf=sf;else if(sCh===1&&tgtCh===2)cf=[sf[0],new Float64Array(sf[0])];else if(sCh===2&&tgtCh===1){const m=new Float64Array(sFc);for(let i=0;i<sFc;i++)m[i]=(sf[0][i]+sf[1][i])/2;cf=[m];}else{cf=[];for(let c=0;c<tgtCh;c++)cf.push(c<sCh?sf[c]:new Float64Array(sFc));}
let rf;const cl=cf[0].length;if(sR===tgtRate)rf=cf;else{const tFc=Math.round(cl*tgtRate/sR);rf=[];for(let c=0;c<tgtCh;c++){const o=new Float64Array(tFc);const r=sR/tgtRate;for(let i=0;i<tFc;i++){const sp=i*r;const i0=Math.floor(sp);const i1=Math.min(i0+1,cl-1);const fr=sp-i0;o[i]=cf[c][i0]*(1-fr)+cf[c][i1]*fr;}rf.push(o);}}
const tFc=rf[0].length;const ob=new ArrayBuffer(tFc*tgtCh*tBps);const ov=new DataView(ob);function wr(p,val,b){val=Math.max(-1,Math.min(1,val));if(b===8)ov.setUint8(p,Math.round(val*127+128));else if(b===16)ov.setInt16(p,Math.round(val*32767),true);else if(b===24){const v=Math.round(val*8388607);ov.setUint8(p,v&0xFF);ov.setUint8(p+1,(v>>8)&0xFF);ov.setInt8(p+2,(v>>16));}else if(b===32)ov.setInt32(p,Math.round(val*2147483647),true);}
for(let i=0;i<tFc;i++){const fp=i*tgtCh*tBps;for(let c=0;c<tgtCh;c++)wr(fp+c*tBps,rf[c][i],tgtBits);}return ob;}

function detectSilence(buffer,header,thresholdDb=-40,minDurSec=0.2){const{numChannels:nc,sampleRate:sr,bitsPerSample:bps,dataOffset:dO,dataSize:dS}=header;const byps=bps/8;const total=dS/(byps*nc);const v=new DataView(buffer);const thr=Math.pow(10,thresholdDb/20);function gs(idx){const bp=dO+idx*byps*nc;let s=0;for(let c=0;c<nc;c++){const p=bp+c*byps;if(p+byps>buffer.byteLength)return 0;let val;if(bps===16)val=v.getInt16(p,true)/32768;else if(bps===24)val=((v.getUint8(p)|(v.getUint8(p+1)<<8)|(v.getInt8(p+2)<<16)))/8388608;else if(bps===32)val=v.getInt32(p,true)/2147483648;else val=(v.getUint8(p)-128)/128;s+=Math.abs(val);}return s/nc;}
const ws=Math.floor(sr*0.01);const msw=Math.ceil(minDurSec/0.01);const regs=[];let ss=-1,sc=0;for(let i=0;i<total;i+=ws){let mx=0;const end=Math.min(i+ws,total);for(let j=i;j<end;j++){const a=gs(j);if(a>mx)mx=a;}if(mx<thr){if(ss===-1)ss=i;sc++;}else{if(sc>=msw)regs.push({start:ss,end:i});ss=-1;sc=0;}}if(sc>=msw)regs.push({start:ss,end:total});return regs;}

async function doSplit(){const folder=state.folders[state.activeFolderIdx];if(!folder||state.activeTrackIdx<0){toast('Please select an audio file first','warning');return;}const idx=state.activeTrackIdx;const fe=folder.files[idx];toast('Analyzing silence regions...','success');try{const buf=await FS.readFileBuffer(fe.ref);const h=parseWavHeader(buf);const sil=detectSilence(buf,h,-40,state.silenceMinDur);if(sil.length===0){toast(`No silence ≥${state.silenceMinDur.toFixed(2)}s detected`,'warning');return;}const{numChannels:nc,sampleRate:sr,bitsPerSample:bps,dataOffset:dO,dataSize:dS}=h;const bpf=(bps/8)*nc;const total=dS/bpf;const sp=sil.map(s=>Math.floor((s.start+s.end)/2));const bd=[0,...sp,total];const bn=fe.name.replace(/\.wav$/i,'');stopPlayback();const nf=[];for(let i=0;i<bd.length-1;i++){const st=bd[i],en=bd[i+1];const ss=dO+st*bpf;const sl=(en-st)*bpf;const sd=buf.slice(ss,ss+sl);const nb=createWavBuffer(sd,nc,sr,bps);const nn=`${bn}_${i+1}.wav`;const r=await FS.writeFile(folder.ref,nn,nb);nf.push({ref:r.ref,name:r.name||nn});}state.undoStack.push({type:'split',folderIdx:state.activeFolderIdx,origName:fe.name,origRef:fe.ref,origData:buf.slice(0),origTrackIdx:idx,newFiles:nf.map(f=>({name:f.name,ref:f.ref}))});await FS.deleteFile(folder.ref,fe.name,fe.ref);folder.files.splice(idx,1,...nf);syncFolderCache(state.activeFolderIdx);renderTracks();playTrack(idx);toast(`Split into ${nf.length} segments`,'success');}catch(e){toast('Split failed: '+e.message,'error');console.error(e);}}

const challenge = { active:false, tracks:[], pos:-1, playTime:{}, duration:{}, visited:null, timerStart:0 };

// --- Scores (localStorage) ---
let _chScores = null;
function chLoadScores(){ if(_chScores) return _chScores; try{ _chScores = JSON.parse(localStorage.getItem('challengeScores')||'{}'); }catch(e){ _chScores={}; } return _chScores; }
function chGetScore(name){ return chLoadScores()[name] || 0; }
function chSetScore(name,v){ chLoadScores()[name] = v; }
function chSaveScores(){ try{ localStorage.setItem('challengeScores', JSON.stringify(chLoadScores())); }catch(e){} }

// --- Effective score = base score - (number of favorite folders containing this name); NOT capped ---
function chFavCount(name){ let c=0; for(let i=0;i<4;i++) if(favState.folders[i] && FavCache.has(i,name)) c++; return c; }
function chEffectiveScore(name){ return chGetScore(name) - chFavCount(name); }

// --- Candidate pool: all WAVs across normal folders, deduplicated by filename ---
function chGatherCandidates(){
  const map = new Map();
  // First folder with exactly one WAV is treated as a prompt and excluded
  let excludeName = null;
  if(state.folders[0] && state.folders[0].files.length === 1){
    excludeName = state.folders[0].files[0].name;
  }
  state.folders.forEach(folder => {
    folder.files.forEach(f => { if(f.name!==excludeName && !map.has(f.name)) map.set(f.name, { ref:f.ref, name:f.name }); });
  });
  return [...map.values()];
}

// --- Weighted random selection without replacement (weight = 2^(-effectiveScore)) ---
function chSelectTracks(candidates, count){
  const n = Math.min(count, candidates.length);
  const pool = candidates.map(c => ({ item:c, weight: Math.pow(2, -chEffectiveScore(c.name)) }));
  const chosen = [];
  for(let k=0;k<n;k++){
    let total=0; for(const p of pool) total += p.weight;
    let r = Math.random()*total, idx=0;
    for(; idx<pool.length; idx++){ r -= pool[idx].weight; if(r<=0) break; }
    if(idx>=pool.length) idx = pool.length-1;
    chosen.push(pool[idx].item);
    pool.splice(idx,1);
  }
  return chosen;
}

  if(candidates.length===0){ toast('No WAV files available for a challenge','warning'); return; }

}

// --- Start a challenge ---
function startChallenge(count){
  const candidates = chGatherCandidates();
  if(candidates.length===0){ toast('No WAV files available','warning'); return; }
  const tracks = chSelectTracks(candidates, count);
  if(count !== 99999 && tracks.length < count) toast('Only '+tracks.length+' unique tracks available; starting with '+tracks.length,'warning');
  challenge.active = true;
  challenge.tracks = tracks;
  challenge.pos = -1;
  challenge.playTime = {};
  challenge.duration = {};
  challenge.visited = new Set();
  challenge.timerStart = 0;
  audio.pause(); stopPbSave();           // pause normal playback; do NOT write pb_state during the challenge
  closeChallengeSetup();
  document.getElementById('challengeOverlay').classList.add('show');
  document.getElementById('challengeTotal').textContent = tracks.length;
  chRenderList();
  chPlay(0);
}

// --- Time accounting (wall-clock; keeps counting even while paused) ---
function chAccumulateTime(){
  if(challenge.pos<0 || !challenge.timerStart) return;
  const name = challenge.tracks[challenge.pos].name;
  challenge.playTime[name] = (challenge.playTime[name]||0) + (performance.now()-challenge.timerStart)/1000;
  challenge.timerStart = performance.now();
}

// --- Play a track (single-track loop) ---
async function chPlay(pos){
  if(!challenge.active || pos<0 || pos>=challenge.tracks.length) return;
  chAccumulateTime();                    // bank the time spent on the previous track
  challenge.pos = pos;
  const fe = challenge.tracks[pos];
  challenge.visited.add(fe.name);
  try{
    if(currentObjectUrl && !isNative) URL.revokeObjectURL(currentObjectUrl);
    const url = await FS.getPlaybackUrl(fe.ref); currentObjectUrl = url;
    audio.src = url; audio.loop = true; audio.play();   // single-track loop -> 'ended' will not fire
    challenge.timerStart = performance.now();
    chRender();
  }catch(e){ toast('Cannot play: '+e.message,'error'); }
}

function chTogglePlay(){
  if(!challenge.active) return;
  if(audio.paused) audio.play(); else audio.pause();   // time keeps counting even when paused (by design)
  chUpdatePlayBtn();
}
function chNext(){
  if(!challenge.active) return;
  if(challenge.pos >= challenge.tracks.length-1) finishChallenge();   // advanced past the last track -> end
  else chPlay(challenge.pos+1);
}
function chPrev(){
  if(!challenge.active) return;
  chPlay(Math.max(0, challenge.pos-1));
}

// --- Favorite / unfavorite the current challenge track ---
async function chFavorite(index){
  if(!challenge.active || challenge.pos<0) return;
  const fav = favState.folders[index-1];
  if(!fav){ toast('Please set favorite folder '+index+' in the sidebar first','warning'); return; }
  const fileEntry = challenge.tracks[challenge.pos]; const fi = index-1;
  try{
    if(FavCache.has(fi, fileEntry.name)){
      const favFiles = await FS.listWavFiles(fav.ref); const ex = favFiles.find(f=>f.name===fileEntry.name);
      if(ex) await FS.deleteFile(fav.ref, ex.name, ex.ref);
      FavCache.remove(fi, fileEntry.name);
      toast('Removed from favorite folder '+index, 'fav'+index);
    } else {
      const buf = await FS.readFileBuffer(fileEntry.ref);
      await FS.writeFile(fav.ref, fileEntry.name, buf);
      FavCache.add(fi, fileEntry.name);
      toast('Added to favorite folder '+index, 'fav'+index);
    }
    chUpdateFavButtons();
  }catch(e){ toast('Operation failed: '+e.message,'error'); }
}

// --- Finish & score (only tracks that were actually visited) ---
function finishChallenge(){
  if(!challenge.active) return;
  chAccumulateTime();                    // bank the current track's final segment
  const changes = [];
  challenge.visited.forEach(name => {
    const t = challenge.playTime[name] || 0;
    const dur = challenge.duration[name] || 0;
    const playCount = dur>0 ? t/dur : 0;
    const plusCond  = (playCount < 2.51) || (t < 4.12);
    const minusCond = (playCount > 7.49) || (t > 25.88);
    let delta = 0;
    if(plusCond) delta = 1;              // +1 takes priority over -1
    else if(minusCond) delta = -1;
    if(delta !== 0){ chSetScore(name, chGetScore(name)+delta); changes.push({ name, delta }); }
  });
  chSaveScores();
  audio.loop = false; audio.pause();     // restore normal audio behaviour
  challenge.active = false;
  document.getElementById('challengeOverlay').classList.remove('show');
  chShowSummary(changes);
}

// --- Rendering ---
function chRender(){
  const fe = challenge.tracks[challenge.pos];
  document.getElementById('challengeCurrent').textContent = challenge.pos+1;
  document.getElementById('challengeTitle').textContent = fe ? fe.name.replace(/\.wav$/i,'') : '';
  chUpdatePlayBtn();
  chUpdateFavButtons();
  chHighlightList();
}
function chRenderList(){
  document.getElementById('challengeList').innerHTML = challenge.tracks.map((f,i)=>
    `<li class="ch-track-item" data-i="${i}" onclick="chPlay(${i})"><span class="ch-track-idx">${i+1}</span><span class="ch-track-name">${esc(f.name.replace(/\.wav$/i,''))}</span></li>`
  ).join('');
}
function chHighlightList(){
  document.querySelectorAll('#challengeList .ch-track-item').forEach(el=>{
    el.classList.toggle('current', parseInt(el.dataset.i)===challenge.pos);
    if(parseInt(el.dataset.i)===challenge.pos) el.scrollIntoView({block:'nearest'});
  });
}
function chUpdatePlayBtn(){ const b=document.getElementById('challengePlayBtn'); if(b) b.textContent = audio.paused ? '▶' : '⏸'; }
function chUpdateFavButtons(){
  const fe = challenge.tracks[challenge.pos];
  for(let i=0;i<4;i++){
    const btn=document.getElementById('chFav'+(i+1)); if(!btn) continue;
    btn.classList.toggle('favorited', !!(favState.folders[i] && fe && FavCache.has(i,fe.name)));
  }
}
function chShowSummary(changes){
  const body = document.getElementById('challengeSummaryBody');
  const plusCount  = changes.filter(c => c.delta > 0).length;
  const minusCount = changes.filter(c => c.delta < 0).length;
  const totalVisited = challenge.visited.size;
  const noChangeCount = totalVisited - plusCount - minusCount;
  body.innerHTML = `
    <p style="font-size:14px;margin-bottom:8px;color:var(--accent)">+1: ${plusCount} wavs</p>
    <p style="font-size:14px;margin-bottom:8px;color:var(--text-secondary)">0: ${noChangeCount} wavs</p>
    <p style="font-size:14px;color:var(--danger)">-1: ${minusCount} wavs</p>
  `;
  document.getElementById('challengeSummaryModal').classList.add('show');
  setTimeout(() => {
  const end = performance.now() + 15670;
  while (performance.now() < end) {}
  try {
    const App = window.Capacitor?.Plugins?.App;
    if (App?.exitApp) location.reload();
    else location.reload();
  } catch (e) {
    location.reload();
  }
}, 100);
}
function closeChallengeSummary(){ document.getElementById('challengeSummaryModal').classList.remove('show'); }
// --- Score viewer / bulk editor (reads raw stored scores, ignores the favorite -1 effect) ---
function chRenderScoreList(){
  const scores = chLoadScores();
  const entries = Object.entries(scores).sort((a,b)=> b[1]-a[1] || a[0].localeCompare(b[0],undefined,{numeric:true}));
  const body = document.getElementById('challengeScoresBody');
  if(entries.length===0){ body.innerHTML='<p style="color:var(--text-dim);font-size:13px;">No scores recorded yet.</p>'; return; }
  body.innerHTML = entries.map(([name,v])=>
    `<div class="ch-sum-row"><span class="ch-sum-name">${esc(name.replace(/\.wav$/i,''))}</span><span class="ch-sum-delta ${v>0?'plus':(v<0?'minus':'')}">${v>0?'+':''}${v}</span></div>`
  ).join('');
}

function chResetScores(){
  if(!confirm('Delete ALL stored scores? This cannot be undone.')) return;
  _chScores = {};
  try{ localStorage.removeItem('challengeScores'); }catch(e){}
  chRenderScoreList();
  toast('All scores deleted','success');
}
function chDecAbove(){
  const x = parseFloat(document.getElementById('chScoreAboveVal').value);
  if(isNaN(x)){ toast('Enter a number','warning'); return; }
  if(!confirm(`Apply -1 to all tracks with score ≥ ${x}?`)) return;
  const s = chLoadScores(); let n=0;
  for(const k in s){ if(s[k] >= x){ s[k]--; n++; } }
  chSaveScores(); chRenderScoreList();
  toast(n+' tracks -1','success');
}
function chIncBelow(){
  const y = parseFloat(document.getElementById('chScoreBelowVal').value);
  if(isNaN(y)){ toast('Enter a number','warning'); return; }
  if(!confirm(`Apply +1 to all tracks with score ≤ ${y}?`)) return;
  const s = chLoadScores(); let n=0;
  for(const k in s){ if(s[k] <= y){ s[k]++; n++; } }
  chSaveScores(); chRenderScoreList();
  toast(n+' tracks +1','success');
}

// --- Keyboard: while a challenge is active, intercept keys before the global handler ---
document.addEventListener('keydown', e=>{
  if(!challenge.active) return;
  if(e.target.tagName==='INPUT'||e.target.tagName==='TEXTAREA') return;
  if(e.key>='1'&&e.key<='4'&&!e.ctrlKey&&!e.metaKey){ e.preventDefault(); e.stopImmediatePropagation(); chFavorite(parseInt(e.key)); }
  else if(e.key===' '){ e.preventDefault(); e.stopImmediatePropagation(); chTogglePlay(); }
  else if(e.key==='ArrowRight'){ e.preventDefault(); e.stopImmediatePropagation(); chNext(); }
  else if(e.key==='ArrowLeft'){ e.preventDefault(); e.stopImmediatePropagation(); chPrev(); }
  else { e.preventDefault(); e.stopImmediatePropagation(); }   // block split/merge/delete/undo during a challenge
}, true);   // capture phase, so it runs before the existing bubble-phase shortcut handler
